Potential NullReferenceException: If no view matches the requested viewType, result remains null, and result.ShowView(viewData) will crash at runtime. This can cause the game to freeze/stuck. 
public ViewBase ShowView(ViewType viewType, ViewData viewData = null)
{
    ViewBase result = null;
    foreach (var view in views)
    {
        if(view.ViewType == viewType)
            result = view;
        else
            view.HideView();
    }
    result.ShowView(viewData);  // CRASH if result is null!
    return result;
}
